// Separate x86 host. The .txt suffix excludes this source from Roadhog's x64 build.
// Calls the verified public ILhObject COM ABI; no registry or binary modifications.
using System;
using System.Collections.Generic;
using System.IO;
using System.Runtime.InteropServices;
using System.Security.Cryptography;
using System.Text;
using System.Web.Script.Serialization;

internal static class LhOcrHost
{
    private const string Prefix = "LH_OCR_JSON:";
    private const string DllHash = "e6b7283f97e75f5aeeb0394fcd9343ae9f2142f1f3ef9d3557b9400dac856c5d";
    private static readonly JavaScriptSerializer Json = new JavaScriptSerializer { MaxJsonLength = 16 * 1024 * 1024 };
    [DllImport("kernel32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
    private static extern IntPtr LoadLibraryEx(string path, IntPtr file, uint flags);
    [DllImport("kernel32.dll", CharSet = CharSet.Ansi, SetLastError = true)]
    private static extern IntPtr GetProcAddress(IntPtr module, string name);
    [DllImport("kernel32.dll")] private static extern bool FreeLibrary(IntPtr module);
    [DllImport("oleaut32.dll")] private static extern int VariantClear(IntPtr value);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int GetClassObject(ref Guid clsid, ref Guid iid, out IntPtr factory);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int CreateInstance(IntPtr self, IntPtr outer, ref Guid iid, out IntPtr instance);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)] private delegate uint Release(IntPtr self);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int Register(IntPtr self, [MarshalAs(UnmanagedType.BStr)] string code, out int result);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int RegisterEx(IntPtr self, [MarshalAs(UnmanagedType.BStr)] string code, [MarshalAs(UnmanagedType.BStr)] string ip, out int result);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int Initialize(IntPtr self, int threads, int gpu,
        [MarshalAs(UnmanagedType.BStr)] string det, [MarshalAs(UnmanagedType.BStr)] string cls,
        [MarshalAs(UnmanagedType.BStr)] string rec, [MarshalAs(UnmanagedType.BStr)] string keys, out int result);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int Detect(IntPtr self, uint image, int length, double confidence, IntPtr details,
        int padding, int maxSide, double boxScore, double boxThreshold, double unclip, int angle, int mostAngle, out IntPtr text);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)] private delegate int Unload(IntPtr self, out int result);

    private static T Method<T>(IntPtr instance, int slot) where T : class
    {
        IntPtr pointer = Marshal.ReadIntPtr(Marshal.ReadIntPtr(instance), checked(slot * IntPtr.Size));
        return (T)(object)Marshal.GetDelegateForFunctionPointer(pointer, typeof(T));
    }
    private static void Emit(object value) { Console.WriteLine(Prefix + Json.Serialize(value)); Console.Out.Flush(); }
    private static string Text(Dictionary<string, object> value, string key, string fallback)
    { object found; return value.TryGetValue(key, out found) && found != null ? Convert.ToString(found) : fallback; }
    private static double Number(Dictionary<string, object> value, string key, double fallback)
    { object found; return value.TryGetValue(key, out found) ? Convert.ToDouble(found) : fallback; }
    private static int Integer(Dictionary<string, object> value, string key, int fallback)
    { double number = Number(value, key, fallback); if (number != Math.Truncate(number)) throw new ArgumentException("Expected integer."); return checked((int)number); }

    [STAThread]
    private static int Main()
    {
        Console.InputEncoding = new UTF8Encoding(false);
        Console.OutputEncoding = new UTF8Encoding(false);
        IntPtr module = IntPtr.Zero, factory = IntPtr.Zero, instance = IntPtr.Zero;
        bool modelLoaded = false;
        string phase = "initialize";
        try
        {
            if (IntPtr.Size != 4) throw new InvalidOperationException("The OCR host must run as x86.");
            string line = Console.ReadLine();
            if (line == null) throw new InvalidOperationException("Missing initialization request.");
            var request = Json.Deserialize<Dictionary<string, object>>(line);
            string dll = Path.GetFullPath(Text(request, "dll_path", ""));
            if (!File.Exists(dll)) throw new FileNotFoundException("Configured OCR DLL does not exist.");
            using (var sha = SHA256.Create())
            using (var stream = File.OpenRead(dll))
            {
                string hash = BitConverter.ToString(sha.ComputeHash(stream)).Replace("-", "").ToLowerInvariant();
                if (hash != DllHash) { Emit(new { status = "error", code = "unsupported_dll_version", phase = phase }); return 2; }
            }
            phase = "load_library";
            module = LoadLibraryEx(dll, IntPtr.Zero, 0x00000100 | 0x00000800);
            if (module == IntPtr.Zero) throw new System.ComponentModel.Win32Exception(Marshal.GetLastWin32Error());
            IntPtr address = GetProcAddress(module, "DllGetClassObject");
            if (address == IntPtr.Zero) throw new InvalidOperationException("Missing DllGetClassObject export.");
            var getClass = (GetClassObject)Marshal.GetDelegateForFunctionPointer(address, typeof(GetClassObject));
            Guid clsid = new Guid("4AA7B9B7-D409-42CE-B4D9-4F215B55B5DF");
            Guid factoryId = new Guid("00000001-0000-0000-C000-000000000046");
            Guid interfaceId = new Guid("C2ECB921-41EE-45E4-9BC0-9CC25DE16894");
            phase = "class_factory";
            Marshal.ThrowExceptionForHR(getClass(ref clsid, ref factoryId, out factory));
            phase = "create_instance";
            Marshal.ThrowExceptionForHR(Method<CreateInstance>(factory, 3)(factory, IntPtr.Zero, ref interfaceId, out instance));
            string registrationCode = Text(request, "registration_code", "");
            if (registrationCode.Length != 0)
            {
                phase = "registration";
                string ip = Text(request, "registration_ip", "");
                int registered;
                int hr = ip.Length == 0 ? Method<Register>(instance, 8)(instance, registrationCode, out registered)
                    : Method<RegisterEx>(instance, 9)(instance, registrationCode, ip, out registered);
                Marshal.ThrowExceptionForHR(hr);
                if (registered != 1) { Emit(new { status = "error", phase = phase, code = "registration_failed", plugin_code = registered }); return 2; }
            }
            request.Remove("registration_code"); registrationCode = null;
            phase = "model_initialization";
            int initialized;
            int threadCount = Integer(request, "threads", 2);
            if (threadCount < 1 || threadCount > 32) throw new ArgumentException("Invalid thread count.");
            Marshal.ThrowExceptionForHR(Method<Initialize>(instance, 17)(instance, threadCount, -1,
                Text(request, "det_path", ""), Text(request, "cls_path", ""), Text(request, "rec_path", ""), Text(request, "keys_path", ""), out initialized));
            if (initialized != 1)
            {
                Emit(new { status = "error", phase = phase, code = initialized == -88 ? "license_required" : "model_initialization_failed", plugin_code = initialized, host_bits = 32 });
                return 2;
            }
            modelLoaded = true;
            Emit(new { status = "ready", engine = "lh.dll", host_bits = 32, dll_sha256 = DllHash });
            while ((line = Console.ReadLine()) != null)
            {
                request = Json.Deserialize<Dictionary<string, object>>(line);
                if (Text(request, "op", "") == "shutdown") break;
                if (Text(request, "op", "") != "recognize") throw new ArgumentException("Unknown operation.");
                phase = "recognize";
                byte[] image = Convert.FromBase64String(Text(request, "image_base64", ""));
                if (image.Length == 0 || image.Length > 8 * 1024 * 1024) throw new ArgumentException("Invalid image size.");
                double confidence = Number(request, "confidence", 0.1), boxScore = Number(request, "box_score", 0.5), boxThreshold = Number(request, "box_threshold", 0.3);
                double unclip = Number(request, "unclip", 1.6);
                int padding = Integer(request, "padding", 50), maxSide = Integer(request, "max_side", 1024);
                if (!(confidence >= 0 && confidence <= 1 && boxScore >= 0 && boxScore <= 1 && boxThreshold >= 0 && boxThreshold <= 1 && unclip > 0 && unclip <= 10)
                    || padding < 0 || padding > 1024 || maxSide < 32 || maxSide > 4096) throw new ArgumentException("Invalid OCR options.");
                GCHandle pinned = GCHandle.Alloc(image, GCHandleType.Pinned);
                IntPtr details = Marshal.AllocCoTaskMem(16), plainText = IntPtr.Zero;
                try
                {
                    Marshal.Copy(new byte[16], 0, details, 16);
                    uint pointer = unchecked((uint)pinned.AddrOfPinnedObject().ToInt32());
                    Marshal.ThrowExceptionForHR(Method<Detect>(instance, 18)(instance, pointer, image.Length, confidence, details,
                        padding, maxSide, boxScore, boxThreshold, unclip, 0, 0, out plainText));
                    string raw = plainText == IntPtr.Zero ? "" : Marshal.PtrToStringBSTR(plainText);
                    int variantType = Marshal.ReadInt16(details) & 0xFFFF;
                    if (variantType != 0 && variantType != 8) throw new InvalidOperationException("Unexpected OCR detail type.");
                    string detailText = variantType == 8 ? (string)Marshal.GetObjectForNativeVariant(details) : "";
                    if (raw == "-88") { Emit(new { status = "error", phase = phase, code = "license_required", plugin_code = -88 }); return 2; }
                    Emit(new { status = "ok", engine = "lh.dll", raw_text = raw, raw_details = detailText });
                }
                finally
                {
                    if (plainText != IntPtr.Zero) Marshal.FreeBSTR(plainText);
                    VariantClear(details); Marshal.FreeCoTaskMem(details); pinned.Free();
                }
            }
            return 0;
        }
        catch (Exception error)
        {
            // Plugin exception messages may contain registration data; retain only type and HRESULT.
            Emit(new { status = "error", phase = phase, code = "host_error", exception_type = error.GetType().FullName, hresult = error.HResult });
            return 3;
        }
        finally
        {
            if (modelLoaded && instance != IntPtr.Zero) { try { int ignored; Method<Unload>(instance, 20)(instance, out ignored); } catch { } }
            if (instance != IntPtr.Zero) Method<Release>(instance, 2)(instance);
            if (factory != IntPtr.Zero) Method<Release>(factory, 2)(factory);
            if (module != IntPtr.Zero) FreeLibrary(module);
        }
    }
}
