判断队友是否有肉体异常

bool PartyMemberHasPhysicalAbnormal(PartyMember *member)
{
    int count = *(int16_t *)((char *)member + 0x77);
    if (count < 0)
        return false;
    if (count > 112)
        count = 112;
    uintptr_t entries =
        (uintptr_t)member + 0x79;
    for (int i = 0; i < count; ++i)
    {
        uintptr_t entry =
            entries + i * 0x12;
        uint32_t category =
            *(uint32_t *)(entry + 0x08);
        if (category == 2)
            return true;
    }
    return false;
}

统计肉体异常数量：

int CountPartyMemberPhysicalAbnormal(PartyMember *member)
{
    int count = *(int16_t *)((char *)member + 0x77);
    count = max(0, min(count, 112));
    int physicalCount = 0;
    for (int i = 0; i < count; ++i)
    {
        uintptr_t entry =
            (uintptr_t)member + 0x79 + i * 0x12;
        if (*(uint32_t *)(entry + 0x08) == 2)
            ++physicalCount;
    }
    return physicalCount;
}

C# 红黑树遍历骨架

下面只演示 内存快照、离线测试程序或已获授权插件 中的数据结构遍历。进程注入、反作弊绕过和在线游戏外部读写实现不展开。

using System;
using System.Collections.Generic;
public interface IReadOnlyMemory
{
    T Read<T>(nuint address) where T : unmanaged;
    nuint ReadPointer(nuint address);
    string ReadUtf16(nuint address, int maxCharacters);
}
public readonly record struct Vector3(float X, float Y, float Z);
public readonly record struct EntityEntry(
    ushort EntityId,
    nuint EntityPointer
);
public static class EntityTree
{
    private const nuint LeftOffset   = 0x00;
    private const nuint ParentOffset = 0x08;
    private const nuint RightOffset  = 0x10;
    private const nuint IsNilOffset  = 0x19;
    private static bool IsNil(IReadOnlyMemory memory, nuint node)
    {
        return node == 0 || memory.Read<byte>(node + IsNilOffset) != 0;
    }
    private static nuint Successor(
        IReadOnlyMemory memory,
        nuint header,
        nuint node)
    {
        nuint right = memory.ReadPointer(node + RightOffset);
        if (!IsNil(memory, right))
        {
            nuint current = right;
            while (true)
            {
                nuint left = memory.ReadPointer(current + LeftOffset);
                if (IsNil(memory, left))
                    return current;
                current = left;
            }
        }
        nuint parent = memory.ReadPointer(node + ParentOffset);
        while (parent != 0 &&
               parent != header &&
               node == memory.ReadPointer(parent + RightOffset))
        {
            node = parent;
            parent = memory.ReadPointer(parent + ParentOffset);
        }
        return parent;
    }
    public static IEnumerable<nuint> EnumerateNodes(
        IReadOnlyMemory memory,
        nuint header,
        int maximumNodes = 65_536)
    {
        if (header == 0)
            yield break;
        // header->left 是最小节点。
        nuint node = memory.ReadPointer(header + LeftOffset);
        var visited = new HashSet<nuint>();
        for (int index = 0;
             index < maximumNodes &&
             node != 0 &&
             node != header;
             index++)
        {
            // 防止结构变化、损坏或读取不同步导致死循环。
            if (!visited.Add(node))
                yield break;
            if (IsNil(memory, node))
                yield break;
            yield return node;
            node = Successor(memory, header, node);
        }
    }
    public static IEnumerable<EntityEntry> EnumerateEntities(
        IReadOnlyMemory memory,
        nuint gameBase)
    {
        const nuint entitySystemGlobalRva = 0x904690;
        const nuint entityMapOffset = 0x58;
        const nuint nodeEntityIdOffset = 0x20;
        const nuint nodeEntityPointerOffset = 0x28;
        nuint entitySystem =
            memory.ReadPointer(gameBase + entitySystemGlobalRva);
        if (entitySystem == 0)
            yield break;
        nuint header =
            memory.ReadPointer(entitySystem + entityMapOffset);
        foreach (nuint node in EnumerateNodes(memory, header))
        {
            ushort entityId =
                memory.Read<ushort>(node + nodeEntityIdOffset);
            nuint entity =
                memory.ReadPointer(node + nodeEntityPointerOffset);
            if (entity != 0)
                yield return new EntityEntry(entityId, entity);
        }
    }
    public static Vector3 ReadEntityPosition(
        IReadOnlyMemory memory,
        nuint entity)
    {
        return new Vector3(
            memory.Read<float>(entity + 0x4E8),
            memory.Read<float>(entity + 0x4EC),
            memory.Read<float>(entity + 0x4F0)
        );
    }
}

周围怪物筛选逻辑

还需要一个：

CEntity* -> CharacterObject*

的解析步骤。

游戏内部对应：

Game.dll + 0x49B8A0 -> sub_18049B8A0

它的行为是：

entity
    -> entity->vfunc_184()
    -> 返回 GameObject/扩展容器
    -> vfunc_80(index = 1, &actor)
    -> CharacterObject*

所以在 C# 的架构上可以先定义一个解析回调：

public static IEnumerable<(nuint Actor, float Distance)> FindNearbyMonsters(
    IReadOnlyMemory memory,
    nuint gameBase,
    Func<nuint, nuint> resolveCharacterObject,
    float radius)
{
    ushort localEntityId =
        memory.Read<ushort>(gameBase + 0xD21798);
    var entities = new List<EntityEntry>(
        EntityTree.EnumerateEntities(memory, gameBase)
    );
    EntityEntry? localEntry = null;
    foreach (EntityEntry entry in entities)
    {
        if (entry.EntityId == localEntityId)
        {
            localEntry = entry;
            break;
        }
    }
    if (localEntry is null)
        yield break;
    Vector3 playerPosition =
        EntityTree.ReadEntityPosition(
            memory,
            localEntry.Value.EntityPointer
        );
    float radiusSquared = radius * radius;
    foreach (EntityEntry entry in entities)
    {
        if (entry.EntityId == localEntityId)
            continue;
        nuint actor =
            resolveCharacterObject(entry.EntityPointer);
        if (actor == 0)
            continue;
        uint objectType =
            memory.Read<uint>(actor + 0x20);
        if (objectType != 2)
            continue;
        Vector3 position =
            EntityTree.ReadEntityPosition(
                memory,
                entry.EntityPointer
            );
        float dx = position.X - playerPosition.X;
        float dy = position.Y - playerPosition.Y;
        float dz = position.Z - playerPosition.Z;
        float distanceSquared =
            dx * dx + dy * dy + dz * dz;
        if (distanceSquared > radiusSquared)
            continue;
        yield return (
            actor,
            MathF.Sqrt(distanceSquared)
        );
    }
}

resolveCharacterObject 在游戏内部就是 sub_18049B8A0 。对于纯外部 C# 读取器，当前导出文件还没有证明一个稳定的“直接指针偏移链”，所以不能诚实地把它写成 entity + 某偏移 。需要再分析 CEntity::vfunc_184 和后面的扩展容器布局；或者在获授权的进程内插件中调用该函数。

5. C# 遍历附近 NPC 示例

下面没有包含进程附加、反作弊绕过或写内存逻辑。 IMemoryReader 代表你自己的只读内存读取层。

using System;
using System.Collections.Generic;
using System.Numerics;
public interface IMemoryReader
{
    byte ReadByte(ulong address);
    ushort ReadUInt16(ulong address);
    ulong ReadUInt64(ulong address);
    float ReadSingle(ulong address);
    // 用于避免读取无效指针。
    bool IsReadable(ulong address, int size);
}
public sealed class NearbyNpc
{
    public ushort EntityId { get; init; }
    public ulong EntityAddress { get; init; }
    public Vector3 Position { get; init; }
    public float Distance { get; init; }
}
public static class EntityScanner
{
    // 主游戏模块 RVA。
    private const ulong EntitySystemPointerRva = 0x904690;
    private const ulong LocalEntityIdRva = 0xD21798;
    // CEntitySystem。
    private const ulong EntityMapHeaderOffset = 0x58;
    // 红黑树节点。
    private const ulong NodeLeftOffset = 0x00;
    private const ulong NodeParentOffset = 0x08;
    private const ulong NodeRightOffset = 0x10;
    private const ulong NodeIsNilOffset = 0x19;
    private const ulong NodeEntityIdOffset = 0x20;
    private const ulong NodeEntityPointerOffset = 0x28;
    // CEntity。
    private const ulong EntityTypeOffset = 0xF2;
    private const ulong EntityPositionOffset = 0x4E8;
    private const ushort EntityTypeNpc = 3;
    public static IReadOnlyList<NearbyNpc> GetNearbyNpcs(
        IMemoryReader memory,
        ulong gameModuleBase,
        float radius)
    {
        var result = new List<NearbyNpc>();
        if (radius <= 0 || !float.IsFinite(radius))
            return result;
        ulong entitySystem =
            memory.ReadUInt64(gameModuleBase + EntitySystemPointerRva);
        if (entitySystem == 0 ||
            !memory.IsReadable(entitySystem + EntityMapHeaderOffset, 8))
        {
            return result;
        }
        ulong header =
            memory.ReadUInt64(entitySystem + EntityMapHeaderOffset);
        if (header == 0 || !memory.IsReadable(header, 0x30))
            return result;
        ushort localEntityId =
            memory.ReadUInt16(gameModuleBase + LocalEntityIdRva);
        ulong localEntity =
            FindEntityById(memory, header, localEntityId);
        if (localEntity == 0 ||
            !memory.IsReadable(localEntity + EntityPositionOffset, 12))
        {
            return result;
        }
        Vector3 playerPosition =
            ReadPosition(memory, localEntity);
        float radiusSquared = radius * radius;
        // MSVC std::map header->left 指向最小节点。
        ulong node = memory.ReadUInt64(header + NodeLeftOffset);
        // 防止树在读取过程中变化而形成死循环。
        const int maxNodes = 65536;
        for (int visited = 0;
             node != 0 && node != header && visited < maxNodes;
             visited++)
        {
            if (!memory.IsReadable(node, 0x30))
                break;
            ushort entityId =
                memory.ReadUInt16(node + NodeEntityIdOffset);
            ulong entity =
                memory.ReadUInt64(node + NodeEntityPointerOffset);
            if (entity != 0 &&
                entityId != localEntityId &&
                memory.IsReadable(entity + EntityPositionOffset, 12))
            {
                ushort entityType =
                    memory.ReadUInt16(entity + EntityTypeOffset);
                if (entityType == EntityTypeNpc)
                {
                    Vector3 position =
                        ReadPosition(memory, entity);
                    float distanceSquared =
                        Vector3.DistanceSquared(
                            playerPosition,
                            position);
                    if (float.IsFinite(distanceSquared) &&
                        distanceSquared <= radiusSquared)
                    {
                        result.Add(new NearbyNpc
                        {
                            EntityId = entityId,
                            EntityAddress = entity,
                            Position = position,
                            Distance = MathF.Sqrt(distanceSquared)
                        });
                    }
                }
            }
            ulong next = GetNextNode(memory, node, header);
            if (next == node)
                break;
            node = next;
        }
        result.Sort(
            static (left, right) =>
                left.Distance.CompareTo(right.Distance));
        return result;
    }
    private static Vector3 ReadPosition(
        IMemoryReader memory,
        ulong entity)
    {
        return new Vector3(
            memory.ReadSingle(entity + EntityPositionOffset + 0),
            memory.ReadSingle(entity + EntityPositionOffset + 4),
            memory.ReadSingle(entity + EntityPositionOffset + 8));
    }
    private static ulong FindEntityById(
        IMemoryReader memory,
        ulong header,
        ushort wantedId)
    {
        if (wantedId == 0)
            return 0;
        // header->parent 是 root。
        ulong node = memory.ReadUInt64(
            header + NodeParentOffset);
        for (int guard = 0;
             node != 0 &&
             node != header &&
             guard < 65536;
             guard++)
        {
            if (!memory.IsReadable(node, 0x30) ||
                IsNil(memory, node))
            {
                return 0;
            }
            ushort nodeId =
                memory.ReadUInt16(node + NodeEntityIdOffset);
            if (wantedId < nodeId)
            {
                node = memory.ReadUInt64(
                    node + NodeLeftOffset);
            }
            else if (wantedId > nodeId)
            {
                node = memory.ReadUInt64(
                    node + NodeRightOffset);
            }
            else
            {
                return memory.ReadUInt64(
                    node + NodeEntityPointerOffset);
            }
        }
        return 0;
    }
    private static ulong GetNextNode(
        IMemoryReader memory,
        ulong node,
        ulong header)
    {
        ulong right =
            memory.ReadUInt64(node + NodeRightOffset);
        // 有右子树：后继是右子树的最左节点。
        if (!IsNil(memory, right))
        {
            ulong current = right;
            for (int guard = 0; guard < 1024; guard++)
            {
                ulong left =
                    memory.ReadUInt64(
                        current + NodeLeftOffset);
                if (IsNil(memory, left))
                    return current;
                current = left;
            }
            return header;
        }
        // 没有右子树：向父节点走。
        ulong parent =
            memory.ReadUInt64(node + NodeParentOffset);
        for (int guard = 0;
             parent != 0 &&
             parent != header &&
             guard < 1024;
             guard++)
        {
            ulong parentRight =
                memory.ReadUInt64(
                    parent + NodeRightOffset);
            if (node != parentRight)
                break;
            node = parent;
            parent = memory.ReadUInt64(
                parent + NodeParentOffset);
        }
        return parent;
    }
    private static bool IsNil(
        IMemoryReader memory,
        ulong node)
    {
        return node == 0 ||

public sealed class ProcessMemory : IDisposable
{
    private const uint ProcessVmRead = 0x0010;
    private const uint ProcessQueryInformation = 0x0400;
    private readonly nint _handle;
    public ProcessMemory(int processId)
    {
        _handle = OpenProcess(
            ProcessVmRead | ProcessQueryInformation,
            false,
            processId);
        if (_handle == 0)
            throw new Win32Exception(Marshal.GetLastWin32Error());
    }
    public byte[] ReadBytes(nint address, int size)
    {
        var buffer = new byte[size];
        if (!ReadProcessMemory(
                _handle,
                address,
                buffer,
                (nuint)size,
                out nuint read) ||
            read != (nuint)size)
        {
            throw new Win32Exception(
                Marshal.GetLastWin32Error(),
                $"Read failed: 0x{address.ToInt64():X}");
        }
        return buffer;
    }
    public byte ReadByte(nint address) =>
        ReadBytes(address, 1)[0];
    public ushort ReadUInt16(nint address) =>
        BitConverter.ToUInt16(ReadBytes(address, 2));
    public uint ReadUInt32(nint address) =>
        BitConverter.ToUInt32(ReadBytes(address, 4));
    public ulong ReadUInt64(nint address) =>
        BitConverter.ToUInt64(ReadBytes(address, 8));
    public float ReadFloat(nint address) =>
        BitConverter.ToSingle(ReadBytes(address, 4));
    public nint ReadPointer(nint address) =>
        (nint)unchecked((long)ReadUInt64(address));
    public void Dispose()
    {
        if (_handle != 0)

    }

public readonly record struct NearbyEntity(
    ushort EntityId,
    uint ServerObjectId,
    nint EntityPointer,
    Vector3 Position,
    float Distance);
public static class NearbyEntityScanner
{
    // Game.dll RVA
    private const long EntitySystemPointerRva = 0x904690;
    private const long ServerObjectTreeRva = 0xD21740;
    private const long LocalEntityIdRva = 0xD21798;
    // CEntitySystem
    private const long EntityTreeOffset = 0x58;
    // Red-black tree node
    private const long LeftOffset = 0x00;
    private const long ParentOffset = 0x08;
    private const long RightOffset = 0x10;
    private const long IsNilOffset = 0x19;
    // Entity-tree node
    private const long EntityNodeIdOffset = 0x20;
    private const long EntityNodePointerOffset = 0x28;
    // Server-object-tree node
    private const long ServerIdOffset = 0x1C;
    private const long ServerEntityIdOffset = 0x20;
    // CEntity position
    private const long PositionXOffset = 0x4E8;
    private const long PositionYOffset = 0x4EC;
    private const long PositionZOffset = 0x4F0;
    private static nint Add(nint address, long offset) =>
        (nint)(address.ToInt64() + offset);
    private static bool IsNil(ProcessMemory memory, nint node)
    {
        return node == 0 ||
               memory.ReadByte(Add(node, IsNilOffset)) != 0;
    }
    private static Vector3 ReadPosition(
        ProcessMemory memory,
        nint entity)
    {
        return new Vector3(
            memory.ReadFloat(Add(entity, PositionXOffset)),
            memory.ReadFloat(Add(entity, PositionYOffset)),
            memory.ReadFloat(Add(entity, PositionZOffset)));
    }
    private static bool IsValidPosition(Vector3 value)
    {
        return float.IsFinite(value.X) &&
               float.IsFinite(value.Y) &&
               float.IsFinite(value.Z) &&
               MathF.Abs(value.X) < 10_000_000f &&
               MathF.Abs(value.Y) < 10_000_000f &&
               MathF.Abs(value.Z) < 10_000_000f;
    }
    // 在 EntitySystem 的 map<ushort, CEntity*> 中查找实体。
    private static nint FindEntity(
        ProcessMemory memory,
        nint entityTreeHeader,
        ushort entityId)
    {
        nint candidate = entityTreeHeader;
        nint node = memory.ReadPointer(
            Add(entityTreeHeader, ParentOffset)); // root
        while (!IsNil(memory, node))
        {
            ushort nodeId = memory.ReadUInt16(
                Add(node, EntityNodeIdOffset));
            if (nodeId >= entityId)
            {
                candidate = node;
                node = memory.ReadPointer(
                    Add(node, LeftOffset));
            }
            else
            {
                node = memory.ReadPointer(
                    Add(node, RightOffset));
            }
        }
        if (candidate == entityTreeHeader)
            return 0;
        ushort candidateId = memory.ReadUInt16(
            Add(candidate, EntityNodeIdOffset));
        if (candidateId != entityId)
            return 0;
        return memory.ReadPointer(
            Add(candidate, EntityNodePointerOffset));
    }
    // std::map 红黑树迭代器的 ++ 操作。
    private static nint NextNode(
        ProcessMemory memory,
        nint header,
        nint node)
    {
        nint right = memory.ReadPointer(
            Add(node, RightOffset));
        if (!IsNil(memory, right))
        {
            nint result = right;
            nint left = memory.ReadPointer(
                Add(result, LeftOffset));
            while (!IsNil(memory, left))
            {
                result = left;
                left = memory.ReadPointer(
                    Add(result, LeftOffset));
            }
            return result;
        }
        nint parent = memory.ReadPointer(
            Add(node, ParentOffset));
        while (parent != header)
        {
            nint parentRight = memory.ReadPointer(
                Add(parent, RightOffset));
            if (node != parentRight)
                break;
            node = parent;
            parent = memory.ReadPointer(
                Add(parent, ParentOffset));
        }
        return parent;
    }
    public static IReadOnlyList<NearbyEntity> Scan(
        ProcessMemory memory,
        nint gameModuleBase,
        float radius)
    {
        if (radius <= 0 || !float.IsFinite(radius))
            throw new ArgumentOutOfRangeException(nameof(radius));
        nint entitySystem = memory.ReadPointer(
            Add(gameModuleBase, EntitySystemPointerRva));
        if (entitySystem == 0)
            return Array.Empty<NearbyEntity>();
        nint entityTreeHeader = memory.ReadPointer(
            Add(entitySystem, EntityTreeOffset));
        if (entityTreeHeader == 0)
            return Array.Empty<NearbyEntity>();
        ushort localEntityId = memory.ReadUInt16(
            Add(gameModuleBase, LocalEntityIdRva));
        if (localEntityId == 0)
            return Array.Empty<NearbyEntity>();
        nint localEntity = FindEntity(
            memory,
            entityTreeHeader,
            localEntityId);
        if (localEntity == 0)
            return Array.Empty<NearbyEntity>();
        Vector3 localPosition =
            ReadPosition(memory, localEntity);
        if (!IsValidPosition(localPosition))
            return Array.Empty<NearbyEntity>();
        nint serverTreeHeader = memory.ReadPointer(
            Add(gameModuleBase, ServerObjectTreeRva));
        if (serverTreeHeader == 0)
            return Array.Empty<NearbyEntity>();
        var result = new List<NearbyEntity>();
        // header->left 是最小节点，也就是 begin()。
        nint node = memory.ReadPointer(
            Add(serverTreeHeader, LeftOffset));
        float radiusSquared = radius * radius;
        // 防止树在读取途中变化造成死循环。
        int guard = 0;
        const int maximumNodes = 100_000;
        while (node != 0 &&
               node != serverTreeHeader &&
               guard++ < maximumNodes)
        {
            nint next;
            try
            {
                next = NextNode(
                    memory,
                    serverTreeHeader,
                    node);
                uint serverObjectId = memory.ReadUInt32(
                    Add(node, ServerIdOffset));
                ushort entityId = memory.ReadUInt16(
                    Add(node, ServerEntityIdOffset));
                if (entityId != 0 &&
                    entityId != localEntityId)
                {
                    nint entity = FindEntity(
                        memory,
                        entityTreeHeader,
                        entityId);
                    if (entity != 0)
                    {
                        Vector3 position =
                            ReadPosition(memory, entity);
                        if (IsValidPosition(position))
                        {
                            float distanceSquared =
                                Vector3.DistanceSquared(
                                    localPosition,
                                    position);
                            if (distanceSquared <= radiusSquared)
                            {
                                result.Add(new NearbyEntity(
                                    entityId,
                                    serverObjectId,
                                    entity,
                                    position,
                                    MathF.Sqrt(distanceSquared)));
                            }
                        }
                    }
                }
            }
            catch (Win32Exception)
            {
                // 游戏可能正好删除或插入节点。
                break;
            }
            node = next;
        }
        return result
            .OrderBy(item => item.Distance)
            .ToArray();
    }
}

建议命名：

常规分支大致等价于：

bool Physics_HasClearLine(Object* from, Object* to, int mode)
{
    Vec3 direction = to->position - from->position;
    if (direction.LengthSquared() < epsilon)
        return true;
    RayHit hit = {};
    IPhysicalWorld* physicsWorld = GetPhysicalWorld();
    int hitCount = physicsWorld->RayWorldIntersection(
        from->position,
        direction,
        257,        // 碰撞对象类别掩码
        47,         // 射线标志
        &hit,
        1,
        nullptr,
        0
    );
    return hitCount == 0;
}

位置来源是：

a2 + 0x18 = 终点 Vec3

返回值：

1 = 两点之间无遮挡
0 = 中间发生碰撞

真正的底层射线函数

sub_1803201B0 返回物理世界对象：

IPhysicalWorld* GetPhysicalWorld()
{
    return gSystem->vfunc_0x168();
}

射线函数位于物理世界对象的虚表：

IPhysicalWorld* world =
    ((IPhysicalWorld* (*)())(gameBase + 0x3201B0))();
uintptr_t vtable = *(uintptr_t*)world;
uintptr_t rayWorldIntersection =
    *(uintptr_t*)(vtable + 0xD0);

所以真正底层函数不是固定的 Game.dll + 某 RVA ，而是：

GetPhysicalWorld() 返回对象
    ↓
对象 vtable + 0xD0
    ↓
RayWorldIntersection

函数入口处大致为：

RCX = IPhysicalWorld*
RDX = 起点 Vec3*
R8 = 方向 Vec3*
R9D = 碰撞类别掩码
EAX = 返回的碰撞数量

判断：

EAX == 0：没有碰撞
EAX > 0 ：射线碰到了物体

为什么能锁定，却不能攻击

锁定系统和攻击有效性是两套逻辑：

锁定目标
    ↓
只保存目标 Entity ID / Server ID
    ↓
允许目标框出现
执行攻击
    ↓
检查距离、角色状态、技能条件
    ↓
检查射线/障碍
    ↓

public interface IReadOnlyMemory
{
    bool TryRead<T>(nuint address, out T value)
        where T : unmanaged;
}
public static class MonsterTargeting
{
    private const int ObjectTypeOffset = 0x20;
    private const int ServerObjectIdOffset = 0x2C;
    private const int TargetServerObjectIdOffset = 0x358;
    private const uint NpcOrMonsterObjectType = 2;
    private static nuint At(nuint address, int offset)
    {
        return address + (nuint)offset;
    }
    public static bool IsMonsterTargetingLocalPlayer(
        IReadOnlyMemory memory,
        nuint monsterActor,
        nuint localPlayerActor)
    {
        if (monsterActor == 0 || localPlayerActor == 0)
            return false;
        if (!memory.TryRead(
                At(monsterActor, ObjectTypeOffset),
                out uint objectType))
        {
            return false;
        }
        if (objectType != NpcOrMonsterObjectType)
            return false;
        if (!memory.TryRead(
                At(localPlayerActor, ServerObjectIdOffset),
                out uint localServerId))
        {
            return false;
        }
        if (localServerId == 0)
            return false;
        if (!memory.TryRead(
                At(monsterActor, TargetServerObjectIdOffset),
                out uint monsterTargetServerId))
        {
            return false;
        }
        return monsterTargetServerId == localServerId;
    }
}

public readonly record struct MonsterAggroInfo(
    nuint ActorAddress,
    uint MonsterServerId,

public static class NearbyMonsterAggro
{
    private const int ObjectTypeOffset = 0x20;
    private const int ServerObjectIdOffset = 0x2C;
    private const int TargetServerObjectIdOffset = 0x358;
    private static nuint At(nuint address, int offset)
    {
        return address + (nuint)offset;
    }
    public static IReadOnlyList<MonsterAggroInfo>
        FindMonstersTargetingLocalPlayer(
            IReadOnlyMemory memory,
            nuint localPlayerActor,
            IEnumerable<nuint> nearbyMonsterActors)
    {
        var result = new List<MonsterAggroInfo>();
        if (localPlayerActor == 0)
            return result;
        if (!memory.TryRead(
                At(localPlayerActor, ServerObjectIdOffset),
                out uint localServerId) ||
            localServerId == 0)
        {
            return result;
        }
        var visited = new HashSet<nuint>();
        foreach (nuint actor in nearbyMonsterActors)
        {
            if (actor == 0 || !visited.Add(actor))
                continue;
            if (!memory.TryRead(
                    At(actor, ObjectTypeOffset),
                    out uint objectType) ||
                objectType != 2)
            {
                continue;
            }
            if (!memory.TryRead(
                    At(actor, ServerObjectIdOffset),
                    out uint monsterServerId))
            {
                continue;
            }
            if (!memory.TryRead(
                    At(actor, TargetServerObjectIdOffset),
                    out uint targetServerId))
            {
                continue;
            }
            if (targetServerId != localServerId)
                continue;
            result.Add(new MonsterAggroInfo(
                actor,
                monsterServerId,
                targetServerId));
        }
        return result;
    }
}

4. “已学习”与“当前可释放”不是同一个概念

最准确的判断流程是：

for each learnedSkill:
    if staticRecord.activationAttribute is manually usable:
        if CanUseSkillNow(skillId):
            skill is currently castable

游戏已经有现成的高层判断函数：

Game.dll + 0x5F7580

建议命名：

bool CanUseSkillNow(uint32_t skillId);

这个判断函数会综合：

- 当前角色状态；
- 当前地图/区域；
- 是否有目标；
- 目标类型与关系；
- HP、MP、DP、材料；
- 武器装备要求；
- 飞行/地面状态；
- 控制异常；
- 技能冷却；
- 特殊技能条件。
在真正发送技能前，代码会先查询已学习技能，再调用 sub_1805F7580 ，通过后才进入 "Send use skill" 路径，因此它比单纯读冷却字段更可靠。

底层两个辅助函数：

sub_1805F6400 -> IsSkillBlockedOrUnusable
sub_1803930C0 -> GetSkillCooldownOrChargeState

通常逻辑是：

bool CanUseSkillNow(uint32_t id)
{
    if (IsSkillBlockedOrUnusable(id))
        return false;
    return GetSkillCooldownOrChargeState(id) != 0;
}

具体内部还有特殊 ID 分支，所以不要只靠这一行伪代码完全替代原函数。

5. C# 只读遍历已学习技能

下面只负责 读取并列出技能 ，不调用游戏技能、不修改内存。

using System;
using System.Collections.Generic;
public interface IMemoryReader
{
    byte ReadByte(nuint address);
    ushort ReadUInt16(nuint address);
    uint ReadUInt32(nuint address);
    ulong ReadUInt64(nuint address);
    nuint ReadPointer(nuint address);
    string ReadUtf16(nuint address, int characterCount);
}
public readonly record struct LearnedSkillSnapshot(
    uint SkillId,
    ushort Level,
    nuint SkillItemAddress,
    string Name,
    uint ToggleState,
    uint CooldownDuration,
    uint CooldownEndTime,
    uint RuntimeState,
    uint SourceFlags
);
public static class LearnedSkillReader
{
    // Game.dll RVA
    private const nuint SkillManagerGlobalRva = 0xD004A0;
    // SkillManager
    private const nuint LearnedSkillTreeOffset = 0x828;
    // MSVC RB tree node
    private const nuint LeftOffset = 0x00;
    private const nuint ParentOffset = 0x08;
    private const nuint RightOffset = 0x10;
    private const nuint IsNilOffset = 0x19;
    // Outer node
    private const nuint OuterSkillIdOffset = 0x20;
    private const nuint OuterLevelTreeOffset = 0x28;
    // Inner level node
    private const nuint InnerLevelOffset = 0x20;
    private const nuint InnerItemListOffset = 0x28;
    // std::list node
    private const nuint ListNextOffset = 0x00;
    private const nuint ListPrevOffset = 0x08;
    private const nuint ListValueOffset = 0x10;
    // SkillItem
    private const nuint ItemSkillIdOffset = 0x08;
    private const nuint ItemNameOffset = 0x18;
    private const nuint ItemCooldownDurationOffset = 0x50;
    private const nuint ItemCooldownEndOffset = 0x54;
    private const nuint ItemToggleStateOffset = 0x60;
    private const nuint ItemLevelOffset = 0x64;
    private const nuint ItemRuntimeStateOffset = 0x6C;
    private const nuint ItemSourceFlagsOffset = 0x74;
    private static bool IsNil(
        IMemoryReader memory,
        nuint node)
    {
        return node == 0 ||
               memory.ReadByte(node + IsNilOffset) != 0;
    }
    private static nuint GetNextTreeNode(
        IMemoryReader memory,
        nuint header,
        nuint node)
    {
        nuint right =
            memory.ReadPointer(node + RightOffset);
        if (!IsNil(memory, right))
        {
            nuint current = right;
            while (true)
            {
                nuint left =
                    memory.ReadPointer(current + LeftOffset);
                if (IsNil(memory, left))
                    return current;
                current = left;
            }
        }
        nuint parent =
            memory.ReadPointer(node + ParentOffset);
        while (parent != 0 &&
               parent != header &&
               node == memory.ReadPointer(parent + RightOffset))
        {
            node = parent;
            parent = memory.ReadPointer(parent + ParentOffset);
        }
        return parent;
    }
    private static IEnumerable<nuint> EnumerateTree(
        IMemoryReader memory,
        nuint header)
    {
        if (header == 0)
            yield break;
        nuint node =
            memory.ReadPointer(header + LeftOffset);
        var visited = new HashSet<nuint>();
        for (int guard = 0;
             node != 0 &&
             node != header &&
             guard < 65_536;
             guard++)
        {
            if (!visited.Add(node))
                yield break;
            if (IsNil(memory, node))
                yield break;
            yield return node;
            node = GetNextTreeNode(
                memory,
                header,
                node);
        }
    }
    private static string ReadMsvcWString(
        IMemoryReader memory,
        nuint stringObject)
    {
        ulong length =
            memory.ReadUInt64(stringObject + 0x10);
        ulong capacity =
            memory.ReadUInt64(stringObject + 0x18);
        if (length == 0)
            return string.Empty;
        // 防止结构变化或读取错误。
        if (length > 256)
            length = 256;
        nuint characters =
            capacity < 8
                ? stringObject
                : memory.ReadPointer(stringObject);
        if (characters == 0)
            return string.Empty;
        return memory.ReadUtf16(
            characters,
            checked((int)length));
    }
    public static IReadOnlyList<LearnedSkillSnapshot>
        EnumerateHighestLearnedSkills(
            IMemoryReader memory,
            nuint gameBase)
    {
        var result =
            new List<LearnedSkillSnapshot>();
        nuint skillManager =
            memory.ReadPointer(
                gameBase + SkillManagerGlobalRva);
        if (skillManager == 0)
            return result;
        nuint outerHeader =
            memory.ReadPointer(
                skillManager + LearnedSkillTreeOffset);
        if (outerHeader == 0)
            return result;
        foreach (nuint outerNode in
                 EnumerateTree(memory, outerHeader))
        {
            uint skillId =
                memory.ReadUInt32(
                    outerNode + OuterSkillIdOffset);
            nuint innerHeader =
                memory.ReadPointer(
                    outerNode + OuterLevelTreeOffset);
            if (innerHeader == 0)
                continue;
            // MSVC tree header->right 是最大 key，也就是最高等级。
            nuint highestLevelNode =
                memory.ReadPointer(
                    innerHeader + RightOffset);
            if (highestLevelNode == 0 ||
                highestLevelNode == innerHeader ||
                IsNil(memory, highestLevelNode))
            {
                continue;
            }
            ushort level =
                memory.ReadUInt16(
                    highestLevelNode + InnerLevelOffset);
            nuint listHeader =
                memory.ReadPointer(
                    highestLevelNode + InnerItemListOffset);
            if (listHeader == 0)
                continue;
            // std::list header->prev 是最后一个节点。
            nuint lastNode =
                memory.ReadPointer(
                    listHeader + ListPrevOffset);
            if (lastNode == 0 ||
                lastNode == listHeader)
            {
                continue;
            }
            nuint item =
                memory.ReadPointer(
                    lastNode + ListValueOffset);
            if (item == 0)
                continue;
            uint itemSkillId =
                memory.ReadUInt32(
                    item + ItemSkillIdOffset);
            // 外层 ID 与 SkillItem ID 不一致时跳过，
            // 防止容器正在被游戏线程修改。
            if (itemSkillId != skillId)
                continue;
            result.Add(new LearnedSkillSnapshot(
                SkillId: skillId,
                Level: level,
                SkillItemAddress: item,
                Name: ReadMsvcWString(
                    memory,
                    item + ItemNameOffset),
                ToggleState: memory.ReadUInt32(
                    item + ItemToggleStateOffset),
                CooldownDuration: memory.ReadUInt32(
                    item + ItemCooldownDurationOffset),
                CooldownEndTime: memory.ReadUInt32(
                    item + ItemCooldownEndOffset),
                RuntimeState: memory.ReadUInt32(
                    item + ItemRuntimeStateOffset),
                SourceFlags: memory.ReadUInt32(
                    item + ItemSourceFlagsOffset)
            ));
        }
        return result;
    }
}

{

    InventoryTreeNode* parent;  // +0x08

单个 InventoryItem 大小：

+0x18 | std::wstring | 显示名称
+0x60 | uint32 | 内部物品类型
+0x74 | uint32 | 装备槽 bitmask；非零通常表示已装备
+0x78 | uint32 | 静态/物品 flags

sub_18038E000 明确证明每页是 27 个槽位：

page = slot / 27;

例如：

using System;
using System.Collections.Generic;
using System.Linq;
public interface IMemoryReader
{
    byte ReadByte(ulong address);
    short ReadInt16(ulong address);
    ushort ReadUInt16(ulong address);
    uint ReadUInt32(ulong address);
    ulong ReadUInt64(ulong address);

    // 读取指定数量的 UTF-16 字符。
    string ReadUtf16(ulong address, int characterCount);
}
public sealed record InventoryItemInfo(
    ulong Address,
    uint InstanceId,
    uint TemplateId,
    long Count,
    string Name,
    string CustomName,
    uint ItemType,
    uint EquipmentMask,
    uint Flags,
    short Slot,
    int Page,
    int Cell,
    ulong ExpiryTimeRaw,
    uint DurationSeconds)
{
    public bool IsEquipped => EquipmentMask != 0;
    public bool IsCashItem =>
        (Flags & 0x1000) != 0;
    public bool IsInNormalBag =>
        Slot >= 0 && EquipmentMask == 0;
}
public sealed record InventorySnapshot(
    ulong ManagerAddress,
    uint Capacity,
    ulong TreeItemCount,

public static class InventoryReader
{
    // Game.dll RVA
    private const ulong InventoryManagerGlobalRva = 0xD004A0;
    // InventoryManager
    private const ulong CapacityOffset = 0x774;
    private const ulong ItemTreeHeaderOffset = 0x778;
    private const ulong ItemTreeCountOffset = 0x780;
    private const ulong EquipmentIdsOffset = 0x788;
    // MSVC 红黑树节点
    private const ulong NodeLeftOffset = 0x00;
    private const ulong NodeParentOffset = 0x08;
    private const ulong NodeRightOffset = 0x10;
    private const ulong NodeIsNilOffset = 0x19;
    private const ulong NodeKeyOffset = 0x20;
    private const ulong NodeItemOffset = 0x28;
    // InventoryItem
    private const ulong ItemInstanceIdOffset = 0x08;
    private const ulong ItemTemplateIdOffset = 0x0C;
    private const ulong ItemCountOffset = 0x10;
    private const ulong ItemNameOffset = 0x18;
    private const ulong ItemTypeOffset = 0x60;
    private const ulong ItemEquipmentMaskOffset = 0x74;
    private const ulong ItemFlagsOffset = 0x78;
    private const ulong ItemSlotOffset = 0x4EE;
    private const ulong ItemCustomNameOffset = 0x4F4;
    private const ulong ItemExpiryOffset = 0x530;
    private const ulong ItemDurationOffset = 0x548;
    private static bool IsNil(
        IMemoryReader memory,
        ulong node)
    {
        return node == 0 ||
               memory.ReadByte(node + NodeIsNilOffset) != 0;
    }
    private static string ReadMsvcWString(
        IMemoryReader memory,
        ulong stringObject)
    {
        ulong length =
            memory.ReadUInt64(stringObject + 0x10);
        ulong capacity =
            memory.ReadUInt64(stringObject + 0x18);
        if (length == 0)
            return string.Empty;
        // 防止客户端更新或内存读取异常。
        if (length > 512 || capacity > 1_000_000)
            return string.Empty;
        // MSVC x64 std::wstring：
        // capacity < 8 时，字符保存在对象内部。
        ulong characters =
            capacity < 8
                ? stringObject
                : memory.ReadPointer(stringObject);
        if (characters == 0)
            return string.Empty;
        return memory.ReadUtf16(
            characters,
            checked((int)length));
    }
    private static ulong GetNextNode(
        IMemoryReader memory,
        ulong header,
        ulong node)
    {
        ulong right =
            memory.ReadPointer(node + NodeRightOffset);
        // 有右子树：取右子树最左节点。
        if (!IsNil(memory, right))
        {
            ulong current = right;
            for (int guard = 0; guard < 1024; guard++)
            {
                ulong left =
                    memory.ReadPointer(
                        current + NodeLeftOffset);
                if (IsNil(memory, left))
                    return current;
                current = left;
            }
            return header;
        }
        // 没有右子树：向父节点回溯。
        ulong parent =
            memory.ReadPointer(node + NodeParentOffset);
        for (int guard = 0;
             parent != 0 &&
             parent != header &&
             guard < 1024;
             guard++)
        {
            ulong parentRight =
                memory.ReadPointer(
                    parent + NodeRightOffset);
            if (node != parentRight)
                break;
            node = parent;
            parent = memory.ReadPointer(
                parent + NodeParentOffset);
        }
        return parent;
    }
    private static IEnumerable<ulong> EnumerateNodes(
        IMemoryReader memory,
        ulong header,
        ulong expectedCount)
    {
        if (header == 0)
            yield break;
        // MSVC map 的 header->left 是最小节点。
        ulong node =
            memory.ReadPointer(header + NodeLeftOffset);
        var visited = new HashSet<ulong>();
        ulong maximum =
            Math.Min(expectedCount + 32, 100_000);
        for (ulong index = 0;
             node != 0 &&
             node != header &&
             index < maximum;
             index++)
        {
            if (!visited.Add(node))
                yield break;
            if (IsNil(memory, node))
                yield break;
            yield return node;
            ulong next =
                GetNextNode(memory, header, node);
            if (next == node)
                yield break;
            node = next;
        }
    }
    public static InventorySnapshot ReadInventory(
        IMemoryReader memory,
        ulong gameModuleBase)
    {
        ulong manager =
            memory.ReadPointer(
                gameModuleBase +
                InventoryManagerGlobalRva);
        if (manager == 0)
        {
            return new InventorySnapshot(
                0,
                0,
                0,
                Array.Empty<InventoryItemInfo>());
        }
        uint capacity =
            memory.ReadUInt32(
                manager + CapacityOffset);
        ulong treeCount =
            memory.ReadUInt64(
                manager + ItemTreeCountOffset);
        ulong header =
            memory.ReadPointer(
                manager + ItemTreeHeaderOffset);
        if (header == 0)
        {
            return new InventorySnapshot(
                manager,
                capacity,
                treeCount,
                Array.Empty<InventoryItemInfo>());
        }
        var items =
            new List<InventoryItemInfo>();
        foreach (ulong node in
                 EnumerateNodes(
                     memory,
                     header,
                     treeCount))
        {
            uint nodeInstanceId =
                memory.ReadUInt32(
                    node + NodeKeyOffset);
            ulong item =
                memory.ReadPointer(
                    node + NodeItemOffset);
            if (item == 0)
                continue;
            uint instanceId =
                memory.ReadUInt32(
                    item + ItemInstanceIdOffset);
            // 防止游戏线程刚好在修改树。
            if (instanceId != nodeInstanceId)
                continue;
            uint templateId =
                memory.ReadUInt32(
                    item + ItemTemplateIdOffset);
            long count = unchecked(
                (long)memory.ReadUInt64(
                    item + ItemCountOffset));
            uint itemType =
                memory.ReadUInt32(
                    item + ItemTypeOffset);
            uint equipmentMask =
                memory.ReadUInt32(
                    item + ItemEquipmentMaskOffset);
            uint flags =
                memory.ReadUInt32(
                    item + ItemFlagsOffset);
            short slot =
                memory.ReadInt16(
                    item + ItemSlotOffset);
            int page =
                slot >= 0 ? slot / 27 : -1;
            int cell =
                slot >= 0 ? slot % 27 : -1;
            string name =
                ReadMsvcWString(
                    memory,
                    item + ItemNameOffset);
            string customName =
                memory.ReadUtf16(
                        item + ItemCustomNameOffset,
                        26)
                    .TrimEnd('\0');
            ulong expiry =
                memory.ReadUInt64(
                    item + ItemExpiryOffset);
            uint duration =
                memory.ReadUInt32(
                    item + ItemDurationOffset);
            items.Add(new InventoryItemInfo(
                Address: item,
                InstanceId: instanceId,
                TemplateId: templateId,
                Count: count,
                Name: name,
                CustomName: customName,
                ItemType: itemType,
                EquipmentMask: equipmentMask,
                Flags: flags,
                Slot: slot,
                Page: page,
                Cell: cell,
                ExpiryTimeRaw: expiry,
                DurationSeconds: duration));
        }
        return new InventorySnapshot(
            manager,
            capacity,
            treeCount,
            items);
    }
    public static uint[] ReadEquipmentInstanceIds(
        IMemoryReader memory,
        ulong manager)
    {
        var result = new uint[32];
        for (int index = 0; index < result.Length; index++)
        {
            result[index] =
                memory.ReadUInt32(
                    manager +
                    EquipmentIdsOffset +
                    (ulong)(index * 4));
        }
        return result;
    }
}

public readonly record struct BagCell(
    short Slot,
    int Page,
    int IndexInPage,
    int Row,
    int Column,
    float NormalizedX,
    float NormalizedY);
public static class BagCoordinates
{
    public const int SlotsPerPage = 27;
    public static bool TryGetCell(
        short slot,
        int columns,
        out BagCell cell)
    {
        cell = default;
        // 0xFFFF 在 Int16 中就是 -1。
        if (slot < 0)
            return false;
        if (columns <= 0 || SlotsPerPage % columns != 0)
            throw new ArgumentOutOfRangeException(
                nameof(columns),
                "columns 必须能整除 27。");
        int rows = SlotsPerPage / columns;
        int page = slot / SlotsPerPage;
        int index = slot % SlotsPerPage;
        int row = index / columns;
        int column = index % columns;
        // 格子中心相对于整个网格的 0~1 坐标。
        float normalizedX =
            (column + 0.5f) / columns;
        float normalizedY =
            (row + 0.5f) / rows;
        cell = new BagCell(
            Slot: slot,
            Page: page,
            IndexInPage: index,
            Row: row,
            Column: column,
            NormalizedX: normalizedX,
            NormalizedY: normalizedY);
        return true;
    }
}

public static class LocalPlayerReader
{
    // Game.dll RVA
    private const nuint EntitySystemPointerRva = 0x904690;
    private const nuint LocalEntityIdRva = 0xD21798;
    private const nuint CurrentMaxHpRva = 0xD267DC;
    private const nuint CurrentHpRva    = 0xD267E0;
    private const nuint CurrentMaxMpRva = 0xD267E4;
    private const nuint CurrentMpRva    = 0xD267E8;
    private const nuint CurrentDpRva    = 0xD267EE;
    private const nuint CameraPitchRva = 0xD1AD14;
    private const nuint CameraRollRva  = 0xD1AD18;
    private const nuint CameraYawRva   = 0xD1AD1C;
    // CEntitySystem
    private const nuint EntityTreeOffset = 0x58;
    // 红黑树节点
    private const nuint NodeLeftOffset   = 0x00;
    private const nuint NodeParentOffset = 0x08;
    private const nuint NodeRightOffset  = 0x10;
    private const nuint NodeIsNilOffset  = 0x19;
    private const nuint NodeIdOffset     = 0x20;
    private const nuint NodeEntityOffset = 0x28;
    // CEntity
    private const nuint PositionOffset = 0x4E8;
    private static nuint ReadPointer(
        IMemoryReader memory,
        nuint address)
    {
        return (nuint)memory.ReadUInt64(address);
    }
    private static bool IsNil(
        IMemoryReader memory,
        nuint node)
    {
        return node == 0 ||
               memory.ReadByte(node + NodeIsNilOffset) != 0;
    }
    private static nuint FindEntityById(
        IMemoryReader memory,
        nuint header,
        ushort entityId)
    {
        nuint node =
            ReadPointer(memory, header + NodeParentOffset);
        for (int guard = 0;
             node != 0 &&
             node != header &&
             guard < 65536;
             guard++)
        {
            if (IsNil(memory, node))
                return 0;
            ushort nodeId =
                memory.ReadUInt16(node + NodeIdOffset);
            if (entityId < nodeId)
            {
                node = ReadPointer(
                    memory,
                    node + NodeLeftOffset);
            }
            else if (entityId > nodeId)
            {
                node = ReadPointer(
                    memory,
                    node + NodeRightOffset);
            }
            else
            {
                return ReadPointer(
                    memory,
                    node + NodeEntityOffset);
            }
        }
        return 0;
    }
    private static float DegreesToRadians(float degrees)
    {
        return degrees * MathF.PI / 180f;
    }
    private static CameraState ReadCamera(
        IMemoryReader memory,
        nuint gameBase)
    {
        float pitch =
            memory.ReadSingle(gameBase + CameraPitchRva);
        float roll =
            memory.ReadSingle(gameBase + CameraRollRva);
        float yaw =
            memory.ReadSingle(gameBase + CameraYawRva);
        float pitchRadians = DegreesToRadians(pitch);
        float yawRadians = DegreesToRadians(yaw);
        // CryEngine 常见坐标：Z 向上，+Y 向前。
        Vector3 horizontalForward = Vector3.Normalize(
            new Vector3(
                -MathF.Sin(yawRadians),
                 MathF.Cos(yawRadians),
                 0));
        Vector3 forward = Vector3.Normalize(
            new Vector3(
                -MathF.Sin(yawRadians) *
                    MathF.Cos(pitchRadians),
                 MathF.Cos(yawRadians) *
                    MathF.Cos(pitchRadians),
                 MathF.Sin(pitchRadians)));
        return new CameraState(
            Pitch: pitch,
            Roll: roll,
            Yaw: yaw,
            HorizontalForward: horizontalForward,
            Forward: forward);
    }
    public static PlayerSnapshot ReadSnapshot(
        IMemoryReader memory,
        nuint gameBase)
    {
        // 同一个 DWORD：
        // 低 16 位是本地 Entity ID，
        // 高 16 位是当前目标 Entity ID。
        ushort localEntityId =
            memory.ReadUInt16(
                gameBase + LocalEntityIdRva);
        ushort targetEntityId =
            memory.ReadUInt16(
                gameBase + LocalEntityIdRva + 2);
        nuint entitySystem =
            ReadPointer(
                memory,
                gameBase + EntitySystemPointerRva);
        nuint localEntity = 0;
        Vector3 position = default;
        if (entitySystem != 0 && localEntityId != 0)
        {
            nuint header =
                ReadPointer(
                    memory,
                    entitySystem + EntityTreeOffset);
            if (header != 0)
            {
                localEntity =
                    FindEntityById(
                        memory,
                        header,
                        localEntityId);
            }
        }
        if (localEntity != 0)
        {
            position = new Vector3(
                memory.ReadSingle(
                    localEntity + PositionOffset + 0),
                memory.ReadSingle(
                    localEntity + PositionOffset + 4),
                memory.ReadSingle(
                    localEntity + PositionOffset + 8));
        }
        var vitals = new PlayerVitals(
            CurrentHp: memory.ReadUInt32(
                gameBase + CurrentHpRva),
            MaxHp: memory.ReadUInt32(
                gameBase + CurrentMaxHpRva),
            CurrentMp: memory.ReadUInt32(
                gameBase + CurrentMpRva),
            MaxMp: memory.ReadUInt32(
                gameBase + CurrentMaxMpRva),
            CurrentDp: memory.ReadUInt16(
                gameBase + CurrentDpRva));
        CameraState camera =
            ReadCamera(memory, gameBase);
        return new PlayerSnapshot(
            EntityId: localEntityId,
            TargetEntityId: targetEntityId,
            EntityAddress: localEntity,
            Position: position,
            Vitals: vitals,
            Camera: camera);
    }
}

关于 Forward 方向

角度地址是高置信度的，但 Forward 的正负方向取决于游戏最终采用的坐标约定。第一次验证时：

1. 相机朝正北记录 Yaw；
2. 顺时针旋转 90°；
3. 检查计算出的 X/Y；
4. 如果方向相反，把 sin 的符号翻转。
Yaw 数值本身不受这个问题影响。

验证顺序

1. 观察 GameBase+D267E0，受伤时应变化。
2. 观察 GameBase+D267E8，施放耗蓝技能时应变化。
3. 观察 CEntity+4E8/4EC/4F0，移动时应变化。
4. 观察 GameBase+D1AD14，抬头/低头时应变化。
5. 观察 GameBase+D1AD1C，水平旋转相机时应变化。

这些偏移仅适用于你上传导出文件对应的客户端版本；客户端更新后，需要通过 S_HIT_POINT 、 S_MANA_POINT 、 Player::UpdateCamera 和 CEntity_SetPosition 的代码特征重新定位。
